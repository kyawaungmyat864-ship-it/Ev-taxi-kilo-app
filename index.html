const START_FEE = 7000; const RATE = 1200; let distance = 0; let previous = null; let watchId = null; let active = false; let hasStarted = false; function metersBetween(a, b) { const R = 6371000; const rad = x => x * Math.PI / 180; const dLat = rad(b.latitude - a.latitude); const dLon = rad(b.longitude - a.longitude); const x = Math.sin(dLat / 2) ** 2 + Math.cos(rad(a.latitude)) * Math.cos(rad(b.latitude)) * Math.sin(dLon / 2) ** 2; return R * 2 * Math.atan2(Math.sqrt(x), Math.sqrt(1 - x)); } function update() { const travel = Math.round(distance / 1000 * RATE); const total = hasStarted ? START_FEE + travel : 0; document.getElementById("km").textContent = (distance / 1000).toFixed(2); document.getElementById("travel").textContent = travel.toLocaleString() + " ကျပ်"; document.getElementById("total").textContent = total.toLocaleString() + " ကျပ်"; } function gpsError(error) { const messages = { 1: "GPS ခွင့်ပြုချက် မရပါ။", 2: "GPS တည်နေရာ မရနိုင်သေးပါ။", 3: "GPS အချိန်ကုန်သွားပါပြီ။" }; document.getElementById("gps").textContent = messages[error.code] || "GPS အမှားရှိနေပါတယ်။"; } function startTrip() { if (!navigator.geolocation) { alert("ဒီ Browser မှာ GPS မရနိုင်ပါ။"); return; } if (active) return; active = true; hasStarted = true; previous = null; distance = 0; document.getElementById("start").disabled = true; document.getElementById("stop").disabled = false; document.getElementById("status").textContent = "🟢 ခရီးစဉ် စတင်နေပါပြီ"; document.getElementById("gps").textContent = "GPS တည်နေရာ ရယူနေပါတယ်။"; update(); watchId = navigator.geolocation.watchPosition( position => { if (!active) return; const p = position.coords; if (p.accuracy > 50) { document.getElementById("gps").textContent = "GPS Signal အားနည်းနေပါတယ်။"; return; } const current = { latitude: p.latitude, longitude: p.longitude }; document.getElementById("gps").textContent = "GPS အလုပ်လုပ်နေပါပြီ။ Accuracy: " + Math.round(p.accuracy) + " m"; if (previous === null) { previous = current; return; } const d = metersBetween(previous, current); if (d >= 3 && d <= 150) { distance += d; previous = current; update(); } else if (d > 150) { previous = current; } }, gpsError, { enableHighAccuracy: true, maximumAge: 0, timeout: 20000 } ); } function stopTrip() { if (!active) return; active = false; if (watchId !== null) { navigator.geolocation.clearWatch(watchId); watchId = null; } document.getElementById("start").disabled = false; document.getElementById("stop").disabled = true; document.getElementById("status").textContent = "🔴 ခရီးစဉ် ပြီးဆုံးပါပြီ"; document.getElementById("gps").textContent = "ခရီးစဉ် ရပ်ထားပါပြီ။"; update(); } function resetTrip() { if (active) stopTrip(); distance = 0; previous = null; hasStarted = false; document.getElementById("status").textContent = "ခရီးစဉ် စတင်ရန် အသင့်ဖြစ်ပါပြီ"; document.getElementById("gps").textContent = "GPS: ခွင့်ပြုချက် မရသေးပါ"; update(); } update();
